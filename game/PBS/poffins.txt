# Poffin kinds that cooking can produce.
#   Name        - shown in the Poffin Case and messages
#   Description - shown when checking a poffin
#   Icon        - x,y of the poffin's cell on the future Poffin Case graphic
#   Priority    - when a poffin matches several kinds, the highest wins
#   Foul        - true for the kind made when cooking fails (same berry
#                 twice, or no flavor left). Never matched otherwise.
#                 At least one kind must be Foul.
#   MinLevel    - the poffin's level (strongest flavor) must be at least this
#   MinFlavors  - the poffin must have at least this many flavors above 0
#   MaxFlavors  - the poffin must have at most this many flavors above 0
#   Flavors     - the poffin's flavors above 0, strongest first, must be
#                 exactly these (ties go Spicy, Dry, Sweet, Bitter, Sour)
# Section names are stored on poffins in save data: renaming or removing
# a section turns poffins players already own into the Foul kind.
#-------------------------------
[FOUL]
Name = Foul Poffin
Description = A poffin that was overcooked or made with the same berry twice. It barely raises any condition.
Icon = 7,0
Priority = 0
Foul = true
#-------------------------------
[SUPERMILD]
Name = Super Mild Poffin
Description = An exquisite poffin of the very highest level. It raises conditions enormously.
Icon = 9,0
Priority = 50
MinLevel = 95
#-------------------------------
